# ── Step 5: Application-Level Budget Enforcement ─────────────────────────────
# In addition to gateway rate limits, the app enforces a $0.05 per-call budget.
# This is a second defense layer in app.py:

BUDGET_LIMIT_USD = 0.05
INPUT_COST_PER_TOKEN = 3.0 / 1_000_000   # $3/M input tokens
OUTPUT_COST_PER_TOKEN = 15.0 / 1_000_000  # $15/M output tokens
MAX_OUTPUT_TOKENS = 1024

def estimate_cost(prompt_text: str, max_output: int = MAX_OUTPUT_TOKENS) -> float:
    input_tokens = len(prompt_text) / 4  # ~4 chars per token
    return (input_tokens * INPUT_COST_PER_TOKEN) + (max_output * OUTPUT_COST_PER_TOKEN)
